cmake_minimum_required(VERSION 3.24)
project(TaikoKeyWASAPI VERSION 0.2.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32)
  message(FATAL_ERROR "Windows 10 or later is required")
endif()
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
  add_link_options(-static)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    add_link_options(-Wl,--threads=2)
  endif()
endif()
add_library(converters STATIC src/audio/Converters.c)
target_include_directories(converters PUBLIC third_party)
if(MSVC)
  target_compile_options(converters PRIVATE /w)
else()
  target_compile_options(converters PRIVATE -w)
endif()
add_library(taiko_core STATIC src/config/KeyConfig.cpp src/audio/SoundBank.cpp src/audio/VoiceMixer.cpp)
target_include_directories(taiko_core PUBLIC src)
target_link_libraries(taiko_core PUBLIC converters ole32 uuid ksuser)
add_library(taiko_windows STATIC src/input/RawInput.cpp src/audio/DeviceCatalog.cpp src/audio/WasapiRenderer.cpp src/app/Controller.cpp)
target_link_libraries(taiko_windows PUBLIC taiko_core ole32 uuid avrt propsys user32 shell32)
add_executable(TaikoKeyWASAPI WIN32 src/app/Main.cpp src/app/OriginalMainUi.cpp src/app/App.rc)
set_source_files_properties(src/app/App.rc PROPERTIES OBJECT_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/src/app/app.manifest;${CMAKE_CURRENT_SOURCE_DIR}/src/image/Background.bmp")
target_link_libraries(TaikoKeyWASAPI PRIVATE taiko_windows comctl32 comdlg32 gdi32)
if(MSVC)
  # App.rc already embeds our manifest; do not generate a second resource ID 1.
  target_link_options(TaikoKeyWASAPI PRIVATE /MANIFEST:NO)
endif()
if(MINGW)
  target_link_options(TaikoKeyWASAPI PRIVATE -municode)
endif()
add_custom_command(TARGET TaikoKeyWASAPI POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:TaikoKeyWASAPI>/assets")
enable_testing()
add_executable(core_tests tests/CoreTests.cpp)
add_executable(TaikoLatencyCapture tools/LatencyCapture.cpp)
target_link_libraries(TaikoLatencyCapture PRIVATE taiko_windows)
if(MINGW)
  target_link_options(TaikoLatencyCapture PRIVATE -municode)
endif()
target_link_libraries(core_tests PRIVATE taiko_core)
target_compile_definitions(core_tests PRIVATE TAIKO_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
add_test(NAME hardware_independent COMMAND core_tests)
set_tests_properties(hardware_independent PROPERTIES TIMEOUT 60)
install(TARGETS TaikoKeyWASAPI RUNTIME DESTINATION .)
install(DIRECTORY assets DESTINATION .)
install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION .)
install(FILES docs/KEY_NAMES.md DESTINATION docs)
install(FILES third_party/VERSIONS.txt third_party/dr_wav.h third_party/miniaudio.h DESTINATION third_party)
install(FILES third_party/LLVM-MINGW-LICENSE.txt DESTINATION third_party)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "TaikoKeyWASAPI-${PROJECT_VERSION}-windows-x64")
include(CPack)
